**数据模型（伪 TS）：**

interface Account { id: string; platform: string; nickname: string; createdAt: number }
interface Content {
  id: string; accountId: string; title: string;

}

  id: string; contentId: string; capturedAt: number;

  revenueCents: number;         // 分成+商单收益合计(分)
  note: string
}

**四维归因（伪代码，可单测）：**
```ts
function attribute(contents: Content[], metrics: MetricSnapshot[], dim: 'tag'|'format'|'platform'|'slot'): AttributionRow[] {
  const bucket = new Map<string, { count:number; views:number; cents:number; hours:number }>()
  for (const c of contents) {
    const ms = metrics.filter(m => m.contentId === c.id)
    const revenue = ms.reduce((s,m)=>s+m.revenueCents, 0)
    const views = ms.reduce((s,m)=>s+m.views, 0)
    const keys = dim === 'tag' ? c.tags : dim === 'format' ? [FORMAT_LABEL[c.format]]
               : dim === 'platform' ? [accountPlatform(c)] : [slotOf(c.publishedAt)]
    for (const k of keys) {
      const b = bucket.get(k) ?? {count:0,views:0,cents:0,hours:0}
      b.count++; b.views += views; b.cents += revenue; b.hours += c.hours
      bucket.set(k, b)
    }
  }
  return [...bucket.entries()].map(([label,b]) => ({
    label, count: b.count, views: b.views,
    revenueYuan: b.cents/100,
    perContentYuan: b.count ? b.cents/100/b.count : 0,
    roiYuanPerHour: b.hours > 0 ? (b.cents/100)/b.hours : null   // 工时 0 回退 null
  })).sort((a,b)=>b.revenueYuan-a.revenueYuan)
}
function slotOf(t:number): string { const h=new Date(t).getHours(); ... 6 段桶 ... }
```

**CSV 导入（伪代码）：**
```ts
async function importCsv(file: File): Promise<{ rows: Content[]; skipped: number }> {
  const { text } = decodeText(await file.arrayBuffer())
  const grid = parseCsv(text)
  const header = grid[0]
  const iTitle = findColumn(header, ['标题','内容','title'])
  const iPlat  = findColumn(header, ['平台','账号','platform'])
  const iDate  = findColumn(header, ['发布时间','日期','date'])
  const iRev   = findColumn(header, ['收益','收入','revenue'])
  // 缺必需列(iTitle/iPlat)直接报错返回，不写入
  const rows: Content[] = []; let skipped = 0
  for (const line of grid.slice(1)) {
    const date = parseFlexibleDate(line[iDate] ?? '')
    if (!line[iTitle] || !date) { skipped++; continue }
    rows.push({ /* 映射字段，收益进对应 metric 或暂存 content.note */ })
  }
  return { rows, skipped }   // 导入前先预览 rows.length/skipped，用户确认再批量 put
}
```
