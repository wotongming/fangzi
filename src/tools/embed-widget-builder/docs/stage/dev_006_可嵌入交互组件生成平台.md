## 技术选型落地

- 产物：Web Component（W3C 标准，Shadow DOM + CSS 变量），任意技术栈站点可用；打包 fflate（MIT）可选；数据本地 + 可选端点；全部动态 import；零成本、商用友好。

## 模块设计

见 architect_006 系统架构；templates（schema+模板）→ config（配置模型）→ gen（生成器）→ snippet（嵌入代码）→ data（回流）；index.vue 五步向导：选模板→配置→预览→嵌入代码→数据面板。

## 关键实现方法

```ts
// 生成器核心（gen.ts，示意）
export function widgetSource(cfg: WidgetConfig): string {
  return `class ${cfg.tagClass} extends HTMLElement {
  connectedCallback(){ this.attachShadow({mode:'open'}); this.render(); this.bindEvents() }
  render(){ this.shadowRoot.innerHTML = \`<style>${themeCss(cfg.theme)}</style>${formHtml(cfg.fields)}\` }
  bindEvents(){ /* 收集数据 → 本地存储 / POST endpoint */ }
}
customElements.define('${cfg.tagName}', ${cfg.tagClass})`
}
// 嵌入代码（snippet.ts）
export function embedSnippet(id: string, ver: string) {
  return `<script src="https://<site>/widgets/${ver}/${id}.js" defer></script>\n<widget-x data-key="${id}"></widget-x>`
}
```
